به نام خداوند جان و خرد
کزین برتر اندیشه برنگذرد
خداوند نام و خداوند جای
خداوند روزی‌ده و رهنمای
خداوند کیوان و گردان سپهر
فروزنده ماه و ناهید و مهر
ز نام و نشان و گمان برتر است
نگارنده بر شده پیکر است
به بینندگان آفریننده را
نبینی مرنجان دو بیننده را
نیابد بدو نیز اندیشه راه
که او برتر از نام و از جایگاه
سخن هر چه زین گوهران بگذرد
نیابد بدو راه جان و خرد
خرد گر سخن برگزیند همی
همان را گزیند که بیند همی
ستودن نداند کس او را چو هست
میان بندگی را ببایدت بست
خرد را و جان را همی سنجد اوی
در اندیشهٔ سخته کی گنجد اوی
بدین آلت رای و جان و زبان
ستود آفریننده را کی توان
به هستیش باید که خستو شوی
ز گفتار بیکار یکسو شوی
پرستنده باشی و جوینده راه
به ژرفی به فرمانش کردن نگاه
توانا بود هر که دانا بود
ز دانش دل پیر برنا بود
از این پرده بر تر سخن‌گاه نیست
ز هستی مر اندیشه را راه نیست
# === گفتار اندر آفرینش عالم ===
از آغاز باید که دانی درست
سر مایهٔ گوهران از نخست
که یزدان ز ناچیز چیز آفرید
بدان تا توانایی آرد پدید
سر مایهٔ گوهران این چهار
برآورده بی‌رنج و بی‌روزگار
یکی آتشی برشده تابناک
میان آب و باد از بر تیره خاک
نخستین که آتش به جنبش دمید
ز گرمیش خشکی پدید آمدید
وز آن پس ز آرام سردی نمود
ز سردی همان باز تری فزود
چو این چار گوهر به جای آمدند
ز بهر سپنجی سرای آمدند
گهرها یک اندر دگر ساخته
ز هر گونه گردن برافراخته
پدید آمد این گنبد تیزرو
شگفتی نمایندهٔ نو به نو
# === گفتار اندر آفرینش مردم ===
چو زین بگذری مردم آمد پدید
شد این بندها را سراسر کلید
سرش راست بر شد چو سرو بلند
به گفتار خوب و خرد کاربند
پذیرندهٔ هوش و رای و خرد
مر او را دد و دام فرمان برد
ز راه خرد بنگری اندکی
که مردم به معنی چه باشد یکی
مر او را ز هر چیز بهره بُدست
ز نیکو سرشتش همه خیره دست
# === کیومرث ===
پژوهندهٔ نامهٔ باستان
که از پهلوانان زند داستان
چنین گفت کآیین تخت و کلاه
کیومرث آورد و او بود شاه
چو آمد به برج حمل آفتاب
جهان گشت با فر و آیین و تاب
بتابید ازان سان ز برج بره
که گیتی جوان گشت ازان یکسره
کیومرث شد بر جهان کدخدای
نخستین به کوه اندرون ساخت جای
سر بخت و تختش برآمد به کوه
پلنگینه پوشید خود با گروه
ازو اندر آمد هنر پیشه‌گی
بدو مردمی بُد ز بیشه نشین
# === هوشنگ ===
جهاندار هوشنگ با رای و داد
به جای نیا تاج بر سر نهاد
بگشت از برش چرخ سالی چهل
پر از هوش مغز و پر از رای دل
چو بنشست بر جایگاه مهی
چنین گفت بر تخت شاهنشهی
من آنم که بر هفت‌کشور منم
بر آنم که بر داد و بخشش تنم
و زان پس جهان یکسر آباد کرد
همه روی گیتی پر از داد کرد
نخستین یکی گوهر آمد به چنگ
به دانش ز آهن جدا کرد سنگ
سر مایه کرد آهن آبگون
کز آن سنگ خارا کشیدش برون
چو بشناختش کارها ساز کرد
بیاموخت و جنگ‌افزار باز کرد
# === طهمورث ===
پسر بُد مر او را یکی هوشمند
گرانمایه طهمورث دیوبند
همه روز بندد به بند گران
ز دیوان و از جادوان بد کنان
بیامد به تخت پدر بر نشست
کمر بست و بگرفت گرز از نشست
همه دد و دام جهان بنده کرد
به رزم و به داد و به بخشش بلند
# === جمشید ===
جمشید فرزانه بر تخت شد
جهان از بدی‌ها همه بخت شد
منم گفت با فرهٔ ایزدی
همم شهریاری همم موبدی
بدان را ز بد دست کوته کنم
روان را سوی روشنی ره کنم
نخست آلت جنگ را دست برد
در نام جستن بزرگی شمرد
ز کتان و ابریشم و موی قز
قصب کرد پرمایه دیبا و خز
بیاموختشان رشتن و تافتن
به تار اندرون پود را بافتن
چو شد ساخته کار جنگ و شکار
در آمد به بخشیدن روزگار
ز هر پیشه‌ای انجمن ساز کرد
بدین اندرون سال‌ها باز کرد
# === ضحاک ===
یکی مرد بود اندر آن روزگار
ز دشت سواران نیزه‌گذار
گرانمایه هم شاه و هم نیک‌مرد
ز ترس جهاندار با باد سرد
مر آن را پدر مرداس نام بود
که بیدار دل بود و نیکی نمود
پسر را بُد ابلیس دل پر فریب
مر او را نبُد اندر آن شهر عیب
جهان را فریبنده آمد به پیش
ز بیداد کردن فزون شد ز بیش
مر او را ضحاک خواندی به نام
ز بیداد او شد جهان تیره‌فام
دو ماری برآمد ز دوشش پدید
که هر روز خون جوانان چشید
هر آن کو ز شاهان جهان یافتی
به خون دو جوان مارها تافتی
# === کاوه آهنگر ===
خروشید و زد دست بر سر ز شاه
که شاها منم کاوه دادخواه
یکی کاوه آهنگر آمد به دشت
ز شاه ستمگر خروشان گذشت
بر آن چرم آهنگران دست ساخت
درفشی بزرگ از میان برفراخت
خروشان همی رفت در کوی و بر
که ای نامداران پرخاشخر
همی رفت پویان و کاوه دمان
به هر بر ز شهری شدی بی‌گمان
بپیوست چندی بر آن بر سپاه
سوی لشکر آمد به فرمان شاه
# === فریدون ===
فریدون چو شد بر جهان کامگار
ندانست جز خویشتن شهریار
به بند اندر آورد ضحاک را
بخست آن بد اندیش ناپاک را
به کوه دماوند کردش نگون
به بند گران بسته افکند خون
جهان شد پر از داد و آئین و فر
همه رفت بیداد و ظلم از نظر
فریدون فرخ فرشته نبود
ز مشک و ز عنبر سرشته نبود
به داد و دهش یافت آن نیکویی
تو داد و دهش کن فریدون تویی
# === زال و سیمرغ ===
دستان زال آمد به گیتی پدید
سپیدش سر و موی چون بید سپید
پدرش سام نریمان ز غم تیره شد
ز فرزند با موی سپید خیره شد
بفرمود تا بر زمینش نهند
در آن کوه البرز جایش دهند
سیمرغ بدید آن تن خرد را
از آن پس بپرورد آن مرد را
# === رستم ===
چو رستم ز مادر بزاد اندر آن
جهان شد پر آوازهٔ پهلوان
تنش چون ز پیلان و بالای سرو
میانش چو شیر و به رفتن تذرو
بنیروی بازوی و با فر و زور
نه دیو و نه شیر و نه پیل از او دور
# === رستم و رخش ===
یکی اسب دید اندر آن دشت و راغ
که بود اندر آن گله چون برق و باغ
بر آن اسب بنشست رستم دمان
همی تاخت بر دشت و کوه و کمان
بدو گفت نامش چه باشد بگوی
بگفتند رخش است نام این نکوی
# === رستم و سهراب ===
خبر شد که سهراب جنگی سوار
همی آید از توران با سپاه و تبار
ندانست رستم که سهراب اوست
که فرزند او بود و خون از پوست
دو لشکر برابر فرود آمدند
دو جنگی به میدان فرود آمدند
بکشتی گرفتند مردان جنگ
یکی تیز خنجر یکی تیغ و سنگ
سرانجام رستم بزد تیغ تیز
بدرید پهلوی سهراب ریز
چو دانست سهراب فرزند اوست
خروشان و نالان و زارش بگفت
که من آن زمان چون بدانستمت
کمربند بگشود و بنمودمت
# === بیژن و منیژه ===
یکی داستان است پر آب چشم
دل نازک از رستم آید به خشم
بدان روزگاران یکی شهر بود
پر از مهر و آزادگی بهر بود
بیژن دلاور سوار جوان
که بود اندر ایران یکی پهلوان
به توران زمین رفت و منیژه دید
دل و جانش از مهر او برتپید
# === رستم و اسفندیار ===
یکی داستان است از روزگار
ز رستم چو خوانی و اسفندیار
دو مرد گرانمایه و نامدار
یکی پهلوان و یکی شهریار
بدان رزمگه هر دو صف برکشید
دو لشکر ز جنگ دلیران چکید
# === سیاوش ===
سیاوش که بود اندر ایران زمین
جوانمرد و پاکیزه با آفرین
ز بیداد و ظلم کسی بد ندید
همه نیکویی بود از او پدید
به توران زمین رفت از ایران برون
ز بی‌مهری گشت دلش پر ز خون
# === کیکاووس ===
کیکاووس شاه جهاندار بود
ولی بی‌خرد بود و بیکار بود
به مازندران لشکر آورد شاه
ز بیدانشی تیره شد بختش و راه
گرفتار شد شاه در دست دیو
خروشان و نالان و بی‌یار و نیو
# === هفت‌خوان رستم ===
تهمتن کمر بست و آمد به راه
به یاری شاه و نجات سپاه
نخستین به شیر ژیان برگذشت
دوم جنگ با اژدها بود سخت
سیم با زن جادو درآویخت
چهارم ره تیره بیراه رفت
در پنجمین روز با اولاد جنگ کرد
ششم ارژنگ دیو را کشت مرد
به هفتم رسید و دیو سپید
بکشتش و دادش به ایران امید
# === پایان موقت ===
بسی رنج بردم در این سال سی
عجم زنده کردم بدین پارسی
نمیرم از این پس که من زنده‌ام
که تخم سخن را پراکنده‌ام
هر آن کس که دارد هش و رای و دین
پس از مرگ بر من کند آفرین
